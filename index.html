const SPREADSHEET_ID = '1Eo_6WvsId6l9Wf3CNguBn8McE075XmSwUgTjz2iD4OM';
const DRIVE_FOLDER_ID = '1iHzv36CncuvlEYa69EIMHnLfYuHV7GHL';
const ADMIN_PIN = '808013';

function doGet(e) {
  try {
    if (e && e.parameter && e.parameter.action) {
      let data = {};
      if (e.parameter.data) {
        try { data = JSON.parse(e.parameter.data); } catch(err) { data = {}; }
      }
      const result = handleAction(e.parameter.action, data);
      return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
    }
  } catch(error) {
    return ContentService.createTextOutput(JSON.stringify({success: false, error: error.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
  return HtmlService.createHtmlOutput('
CreatorHub Apps Script Backend Active
')
      .setTitle('CreatorHub - Laporan Harian')
      .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL)
      .addMetaTag('viewport', 'width=device-width, initial-scale=1.0');
}

function doPost(e) {
  const lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    let body = {};
    if (e && e.postData && e.postData.contents) {
      body = JSON.parse(e.postData.contents);
    }
    const action = body.action || (e && e.parameter && e.parameter.action);
    const result = handleAction(action, body);

    return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
  } catch(error) {
    return ContentService.createTextOutput(JSON.stringify({success: false, error: error.toString()})).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function handleAction(action, body) {
  if (action === 'getCreatorsList') return getCreatorsList();
  if (action === 'loginCreator') return loginCreator(body.username, body.pin);
  if (action === 'loginAdmin') return loginAdmin(body.pin);
  if (action === 'getCreatorsData') return getCreatorsData();
  if (action === 'saveCreator') return saveCreator(body.data || body);
  if (action === 'deleteCreator') return deleteCreator(body.rowIdx);
  if (action === 'submitReport') return submitReport(body.data || body);
  if (action === 'getDashboardStats') return getDashboardStats(body.username, body.startDate, body.endDate, body.isAdmin);
  if (action === 'uploadFileServer') return uploadFileServer(body.base64Data, body.filename, body.mimeType);
  return { success: false, error: 'Unknown action: ' + action };
}

function getSpreadsheet() {
  return SpreadsheetApp.openById(SPREADSHEET_ID);
}

function initSheets() {
  const ss = getSpreadsheet();
  let sheetKreator = ss.getSheetByName('Kreator');
  if (!sheetKreator) {
    sheetKreator = ss.insertSheet('Kreator');
    sheetKreator.appendRow(['Nama', 'Username', 'PIN', 'Target', 'WhatsApp', 'Bank', 'TTL', 'Alamat']);
    sheetKreator.appendRow(['Kreator Demo', 'kreator1', '123456', 10, '628123456789', 'BCA 12345', 'Jakarta', 'Jl. Merdeka']);
  }
  let sheetLaporan = ss.getSheetByName('Laporan');
  if (!sheetLaporan) {
    sheetLaporan = ss.insertSheet('Laporan');
    sheetLaporan.appendRow(['Timestamp', 'Tanggal', 'Username', 'Tipe', 'Link_atau_Motif', 'Drive_URL', 'Jumlah']);
  }
}

function getCreatorsList() {
  initSheets();
  const sheet = getSpreadsheet().getSheetByName('Kreator');
  const data = sheet.getDataRange().getValues();
  const creators = [];
  for (let i = 1; i < data.length; i++) {
    if (data[i][0]) {
      creators.push({ name: data[i][0], username: String(data[i][1]) });
    }
  }
  return creators;
}

function loginCreator(username, pin) {
  initSheets();
  const sheet = getSpreadsheet().getSheetByName('Kreator');
  const data = sheet.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][1]).trim() === String(username).trim() && String(data[i][2]).trim() === String(pin).trim()) {
      return { success: true, name: data[i][0], username: data[i][1], target: data[i][3] };
    }
  }
  return { success: false, message: 'Username atau PIN salah!' };
}

function loginAdmin(pin) {
  if (String(pin).trim() === ADMIN_PIN) {
    return { success: true };
  }
  return { success: false, message: 'PIN Admin salah!' };
}

function getCreatorsData() {
  initSheets();
  const sheet = getSpreadsheet().getSheetByName('Kreator');
  const data = sheet.getDataRange().getValues();
  const result = [];
  for (let i = 1; i < data.length; i++) {
    if (data[i][0]) {
      result.push({
        rowIdx: i + 1,
        name: data[i][0],
        username: data[i][1],
        pin: data[i][2],
        target: data[i][3],
        whatsapp: data[i][4],
        bank: data[i][5],
        ttl: data[i][6],
        address: data[i][7]
      });
    }
  }
  return result;
}

function saveCreator(c) {
  initSheets();
  const sheet = getSpreadsheet().getSheetByName('Kreator');
  if (c.rowIdx) {
    sheet.getRange(c.rowIdx, 1, 1, 8).setValues([[c.name, c.username, c.pin, c.target, c.whatsapp, c.bank, c.ttl, c.address]]);
  } else {
    sheet.appendRow([c.name, c.username, c.pin, c.target, c.whatsapp, c.bank, c.ttl, c.address]);
  }
  return { success: true };
}

function deleteCreator(rowIdx) {
  initSheets();
  const sheet = getSpreadsheet().getSheetByName('Kreator');
  sheet.deleteRow(rowIdx);
  return { success: true };
}

function uploadFileServer(base64Data, filename, mimeType) {
  try {
    const folder = DriveApp.getFolderById(DRIVE_FOLDER_ID);
    const decoded = Utilities.base64Decode(base64Data);
    const blob = Utilities.newBlob(decoded, mimeType, filename);
    const file = folder.createFile(blob);
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    return { success: true, url: file.getUrl(), name: filename };
  } catch(e) {
    return { success: false, error: e.toString() };
  }
}

function submitReport(data) {
  initSheets();
  const sheet = getSpreadsheet().getSheetByName('Laporan');
  const now = new Date();
  const dateStr = now.toISOString().split('T')[0];
  let tiktokCount = 0;
  let driveCount = 0;

  if (data.tiktokLinks && data.tiktokLinks.length > 0) {
    data.tiktokLinks.forEach(function(link) {
      sheet.appendRow([now, dateStr, data.username, 'TikTok', link, '', 1]);
      tiktokCount++;
    });
  }

  if (data.driveFiles && data.driveFiles.length > 0) {
    data.driveFiles.forEach(function(file) {
      sheet.appendRow([now, dateStr, data.username, 'Drive', file.name, file.url, 1]);
      driveCount++;
    });
  }

  return {
    success: true,
    username: data.username,
    totalTikTok: tiktokCount,
    totalDrive: driveCount,
    totalAll: tiktokCount + driveCount
  };
}

function getDashboardStats(username, startDate, endDate, isAdmin) {
  initSheets();
  const ss = getSpreadsheet();
  const lapSheet = ss.getSheetByName('Laporan');
  const kreatorSheet = ss.getSheetByName('Kreator');
  
  const lapData = lapSheet.getDataRange().getValues();
  const kreatorData = kreatorSheet.getDataRange().getValues();
  
  if (isAdmin) {
    const creatorsMap = {};
    for (let i = 1; i < kreatorData.length; i++) {
      if (kreatorData[i][0]) {
        creatorsMap[kreatorData[i][1]] = {
          name: kreatorData[i][0],
          username: kreatorData[i][1],
          target: Number(kreatorData[i][3]) || 10,
          achieved: 0,
          tiktok: 0,
          drive: 0
        };
      }
    }

    for (let i = 1; i < lapData.length; i++) {
      const rowDate = lapData[i][1];
      const u = lapData[i][2];
      const type = lapData[i][3];
      
      if (startDate && rowDate < startDate) continue;
      if (endDate && rowDate > endDate) continue;

      if (creatorsMap[u]) {
        creatorsMap[u].achieved += 1;
        if (type === 'TikTok') creatorsMap[u].tiktok += 1;
        if (type === 'Drive') creatorsMap[u].drive += 1;
      }
    }

    return { performanceList: Object.values(creatorsMap) };
  } else {
    let tiktok = 0, drive = 0;
    for (let i = 1; i < lapData.length; i++) {
      const rowDate = lapData[i][1];
      const u = lapData[i][2];
      const type = lapData[i][3];

      if (u === username) {
        if (startDate && rowDate < startDate) continue;
        if (endDate && rowDate > endDate) continue;

        if (type === 'TikTok') tiktok++;
        if (type === 'Drive') drive++;
      }
    }
    return { totalTikTok: tiktok, totalDrive: drive, totalAll: tiktok + drive };
  }
}
